#include<iostream>
#include<algorithm>

using namespace std;

int main(){

    int n,q;
    cin>>n>>q;
    long long p[n];
    for(int i=0;i<n;i++){
        cin>>p[i];
    }
    std::sort(p, p + n, std::greater<long long>());
    long long pre[n+1];
    pre[0]=0;
    for(int i=1; i<=n; i++){
        pre[i]=pre[i-1]+p[i-1];
    }
    
    for(int i=0; i<=n; i++){
        cout<<pre[i]<<" ";
    }

return 0;
}
