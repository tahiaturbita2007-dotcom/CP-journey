sorting in c++

#include<iostream>
#include<algorithm>

using namespace std;

int main(){

    int n,q;
    cin>>n>>q;
    long long p[200005];
    for(int i=0;i<n;i++){
        cin>>p[i];
    }
    std::sort(p, p + n, std::greater<long long>());
    for(int i=0;i<n;i++){
        cout<<p[i]<<" ";
    }
return 0;
}
